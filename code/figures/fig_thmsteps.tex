% fig_thmsteps.tex -- written 2026-11-04 10:25 CET. Theorem "barcode and nearest neighbours" in four steps, on the nine
% points of thm_steps_points.py (which checks every number stated here: p=3 mutual pairs, N-p=6 forest
% edges, p-1=2 bridges (6,8)=1.55 and (3,5)=2.24, max nearest-neighbour distance 0.79). \input inside a figure.
\begin{tikzpicture}[x=1.12cm, y=1.12cm, font=\small,
  pt/.style={circle, fill=black, inner sep=0pt, minimum size=4.4pt},
  nn/.style={-{Stealth[length=5.2pt]}, tabB, line width=1pt, shorten >=2.6pt, shorten <=2.6pt},
  mut/.style={{Stealth[length=5.2pt]}-{Stealth[length=5.2pt]}, tabB, line width=1pt, shorten >=2.6pt, shorten <=2.6pt},
  fe/.style={tabB, line width=1.1pt},
  br/.style={tabV, line width=1.3pt, densely dashed},
  ttl/.style={anchor=south west, align=left, font=\small}]
  \def\pts{\coordinate (1) at (0,0); \coordinate (2) at (0.55,0.35); \coordinate (3) at (1.25,0.15);
    \coordinate (4) at (0.3,1.1); \coordinate (5) at (3.0,1.55); \coordinate (6) at (3.45,1.15);
    \coordinate (7) at (3.78,1.8); \coordinate (8) at (4.3,-0.15); \coordinate (9) at (4.95,0.12);}
  \def\drawpts{\foreach \i in {1,...,9} \node[pt] at (\i) {};}
  % ---- Step 1
  \begin{scope}
    \pts
    \node[ttl] at (-0.35,2.3) {\textbf{1}\enspace Each point points to its nearest\\ neighbour; mutual pairs point both ways};
    \draw[mut] (1) -- (2); \draw[mut] (5) -- (6); \draw[mut] (8) -- (9);
    \draw[nn] (3) -- (2); \draw[nn] (4) -- (2); \draw[nn] (7) -- (6);
    \drawpts
  \end{scope}
  % ---- Step 2
  \begin{scope}[xshift=6.55cm]
    \pts
    \node[ttl] at (-0.35,2.3) {\textbf{2}\enspace Each chain of arrows ends at a\\ mutual pair: $p=3$ clumps, $N-p=6$ edges};
    \draw[tabB!14, line width=15pt, line cap=round, line join=round] (1) -- (2) (3) -- (2) (4) -- (2) (5) -- (6) (7) -- (6) (8) -- (9);
    \draw[fe] (1) -- (2) (3) -- (2) (4) -- (2) (5) -- (6) (7) -- (6) (8) -- (9);
    \drawpts
  \end{scope}
  % ---- Step 3
  \begin{scope}[yshift=-4.05cm]
    \pts
    \node[ttl] at (-0.35,2.3) {\textbf{3}\enspace Grow balls: each clump joins up first;\\ then $p-1=2$ bridges join the clumps};
    \foreach \i in {1,...,9} \fill[black!7] (\i) circle (0.4);
    \foreach \i in {1,...,9} \draw[black!25, line width=1pt] (\i) circle (0.4);
    \draw[fe] (1) -- (2) (3) -- (2) (4) -- (2) (5) -- (6) (7) -- (6) (8) -- (9);
    \draw[br] (3) -- node[above left=-1pt, tabV] {2.24} (5);
    \draw[br] (6) -- node[right=1pt, tabV] {1.55} (8);
    \drawpts
  \end{scope}
  % ---- Step 4: barcode, bars sorted, one shared scale (1 unit = 2cm)
  \begin{scope}[xshift=6.55cm, yshift=-4.05cm, x=2.0cm]
    \node[ttl] at (-0.175,2.3) {\textbf{4}\enspace The $H_0$ bars are the tree's\\ $N-1=8$ edges, sorted};
    \foreach \len [count=\k] in {0.60,0.65,0.70,0.73,0.73,0.79} \draw[tabB, line width=2.1pt] (0,2.05-0.205*\k) -- (\len,2.05-0.205*\k);
    \draw[tabV, line width=2.1pt] (0,2.05-0.205*7) -- (1.55,2.05-0.205*7);
    \draw[tabV, line width=2.1pt] (0,2.05-0.205*8) -- (2.24,2.05-0.205*8);
    \node[anchor=west, tabB, align=left, font=\footnotesize] at (0.85,1.33) {$N-p=6$ nearest-neighbour\\[-2pt] distances};
    \node[anchor=west, tabV, font=\footnotesize] at (1.6,0.62) {$p-1=2$ bridges};
    \draw[black!60, line width=1pt] (0,0.12) -- (2.4,0.12);
    \foreach \v in {0,1,2} \draw[black!60] (\v,0.12) -- (\v,0.05) node[below, font=\footnotesize] {$\v$};
    \draw[black!45, densely dotted] (0.79,0.12) -- (0.79,1.9);
    \node[above, font=\footnotesize, black!70, inner sep=1pt] at (0.79,1.9) {largest $\delta_i=0.79$};
  \end{scope}
\end{tikzpicture}%
