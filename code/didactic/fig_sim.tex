% fig_sim.tex -- The four simulation figures of the appendix, redrawn from
% matplotlib for readability. Defines \simMain (fig:sim), \simDims (fig:simall), \simProj (fig:proj), \simMixed (fig:sim_mixed).
% Data: data/sim/sim_*.dat from code/simulation/make_figs.py (KDEs of the all-50-neighbour features; eps means).
% Density units: a KDE value y is y% of queries per 0.01 of W1; LT_max panels divide by 10 (% per 0.001).
% W1 eps panel uses the same scale as the W1 density panel's x axis, so the plateau reads against it. \input once.
\providecommand{\didpath}{figs_didactic/}%
\definecolor{didV}{HTML}{D55E00}\definecolor{didB}{HTML}{0072B2}\definecolor{simC}{HTML}{009E73}%
\pgfplotsset{simax/.style={scale only axis, width=5.9cm, height=3.3cm, axis x line=bottom, axis y line=left,
    tick label style={font=\scriptsize}, label style={font=\footnotesize}, major tick length=2pt,
    every axis plot/.append style={line width=1pt, mark=none}, clip=false, ymin=0,
    scaled ticks=false, xticklabel style={/pgf/number format/fixed}, yticklabel style={/pgf/number format/fixed}},
  simw/.style={simax, xlabel={$\Wone$, all 50 neighbours}, ylabel={\% of queries per 0.01}},
  siml/.style={simax, xlabel={$\LTmax$, all 50 neighbours}, ylabel={\% of queries per 0.001}, ylabel shift=4pt},  % clear of the tick labels
