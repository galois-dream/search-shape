% fig_vmf.tex -- Appendix figure for the two-meaning statistic: three toy clouds of 50
% unit directions in 3-D, seen from above the sphere (top row), and the same directions after the statistic's registered
% step, centred, projected onto the leading principal directions and renormalised (bottom row; 2 directions here, so the
% points lie on a circle). Bottom points are drawn at radii 1, 1.06, 1.12, 1.18 in turn so that coinciding points stay
% visible. Data and numbers: data/vmf_fig/ from gen_vmf_fig.py (vmf2 from code/realtext/vmf.py). \input from main.tex.
\providecommand{\didpath}{figs_didactic/}%
\input{\didpath data/vmf_fig/vmf_meta.tex}%
\definecolor{didV}{HTML}{D55E00}\definecolor{didB}{HTML}{0072B2}\definecolor{didG}{HTML}{009E73}%
\pgfplotsset{vmfax/.style={scale only axis, width=3.3cm, height=3.3cm, axis equal image, hide axis, clip=false,
  xmin=-1.25, xmax=1.25, ymin=-1.25, ymax=1.25}}%
\begin{tikzpicture}[font=\footnotesize]
  % #1 case, #2 x shift, #3 title
  \def\vmfcol#1#2#3{%
    \begin{axis}[vmfax, at={(#2,3.9cm)}, anchor=south west]
      \draw[black!25, line width=0.5pt] (axis cs:0,0) circle[radius=1];
      \addplot[only marks, mark=*, mark size=1.2pt, didB, fill opacity=0.8, draw opacity=0] table {\didpath data/vmf_fig/vmf_#1_sphere.dat};
    \end{axis}
    \node[anchor=south, align=center] at ({#2+1.65cm}, 7.35cm) {#3};
    \begin{axis}[vmfax, at={(#2,0cm)}, anchor=south west]
      \draw[black!25, line width=0.5pt] (axis cs:0,0) circle[radius=1];
      \addplot[only marks, mark=*, mark size=1.1pt, didB, fill opacity=0.8, draw opacity=0]
        table[x expr={\thisrow{x}*(1+0.06*mod(\coordindex,4))}, y expr={\thisrow{y}*(1+0.06*mod(\coordindex,4))}] {\didpath data/vmf_fig/vmf_#1_proj.dat};
    \end{axis}}
  \vmfcol{one}{0cm}{(a) one meaning}
  \vmfcol{two}{4.4cm}{(b) two meanings}
  \vmfcol{long}{8.8cm}{(c) one \emph{elongated} meaning}
  % row labels
  \node[anchor=east, align=right, text width=2.2cm] at (-0.15cm, 5.55cm) {on the sphere\\{\scriptsize(seen from above)}};
  \node[anchor=east, align=right, text width=2.2cm] at (-0.15cm, 1.65cm) {after centring, projecting and renormalising};
  % verdicts
  \node[anchor=north, align=center] at (1.65cm, -0.1cm) {spread round the circle\\\textcolor{didG}{\textbf{BIC: one}} \checkmark\\{\scriptsize vmf2 $\vmfGapOne$; two in \vmfHitsOne\ of 200}};
  \node[anchor=north, align=center] at (6.05cm, -0.1cm) {two clumps\\\textcolor{didG}{\textbf{BIC: two}} \checkmark\\{\scriptsize vmf2 $\vmfGapTwo$; two in \vmfHitsTwo\ of 200}};
  \node[anchor=north, align=center] at (10.45cm, -0.1cm) {two clumps, from \emph{one} meaning\\\textcolor{didV}{\textbf{BIC: two}} $\times$\\{\scriptsize vmf2 $\vmfGapLong$; two in \vmfHitsLong\ of 200}};
  \draw[-{Latex[length=5pt]}, black!45, line width=0.6pt] (12.55cm, 5.4cm) -- (12.55cm, 2.2cm);
  \node[anchor=west, align=left, text width=2.3cm, font=\scriptsize] at (12.65cm, 3.8cm) {centring puts the two ends of a long cloud on opposite sides; renormalising pushes them apart};
\end{tikzpicture}%
